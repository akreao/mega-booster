//===== rAthena Script =======================================
//= Mega Booster: Lime Evenor, summoned by the Booster Call
//===== Description: =========================================
//= Part of the Ragnarok Offline mod "mega-booster".
//= kRO's Mega Booster promotion (메가 부스터 프로모션, 2022 on).
//=
//= Using a Booster Call (100337) or Premium Booster Call (101473)
//= runs duplicate_dynamic("Lime Evenor#booster"), which puts a
//= private copy of the NPC below next to the player for 60 s
//= (feature.dynamicnpc_timeout). kRO's quest text calls her
//= 라임 이베노르, the Eden Group officer; rAthena names her
//= "Lime Evenor", so this mod does too.
//=
//= Menu (kRO order, from hazyforest; [unverified: menu layout]):
//=  1. Booster Shop: Coin Shop, Currency Exchange, Mission Shop
//=  2. Booster Missions: Lv 1-80 hunts, Lv 80+ missions
//=  3. Booster Enchant (Lv 100+)
//=  4. Use a Level Up Ticket
//=  5. Claim milestone rewards
//=  6. Help, and a warp to the Prontera Convenience Center
//=  7. Upgrade to the Premium Booster Call
//=
//= Sources: findings.md in the project's booster/ folder.
//= GN = gnjoy notice 278 (2026 guide), HF = hazyforest,
//= CL = kRO client (itemInfo / OngoingQuestInfoList).
//============================================================

// Base for the dynamic NPC. Same pattern as rAthena's Shorty
// (npc/re/quests/HelpMeShorty.txt): a real NPC on an unreachable
// cell, duplicated next to the player by the item script.
prt_in,1,1,0	script	Lime Evenor#booster	4_F_EDEN_OFFICER,{
	.@premium = (countitem(101473) > 0);
	if (!.@premium && countitem(100337) < 1) {
		mes "[Lime Evenor]";
		mes "Hm? Only Booster Call holders can use my services.";
		close;
	}
	mes "[Lime Evenor]";
	mes "Lime Evenor of the Eden Group, at your service, " + strcharinfo(0) + ".";
	mes "What can I do for you?";
	next;
	switch (select("Booster Shop", "Booster Missions", "Booster Enchant", "Use a Level Up Ticket", "Claim milestone rewards", "Help", (.@premium ? "" : "Upgrade to the Premium Booster Call"), "Cancel")) {
	case 1: callsub L_Shop, .@premium; end;
	case 2: callsub L_Missions; end;
	case 3: callsub L_Enchant; end;
	case 4: callsub L_LevelUp; end;
	case 5: callsub L_Milestones, .@premium; end;
	case 6: callsub L_Help; end;
	case 7: callsub L_Upgrade; end;
	default: close;
	}

//------------------------------------------------------------
// 1. Shops
//------------------------------------------------------------
L_Shop:
	.@premium = getarg(0);
	mes "[Lime Evenor]";
	mes "Which shop?";
	next;
	switch (select("Booster Coin Shop", "Currency Exchange", "Booster Mission Shop", "Cancel")) {
	case 1:
		// GN tab 1-3: the Premium Call has its own, cheaper list.
		// [unverified: one source (GN)]
		.@shop$ = (.@premium ? "booster_coin_shop_p" : "booster_coin_shop");
		close2;
		callshop .@shop$, 1;
		end;
	case 2:
		callsub L_Exchange;
		end;
	case 3:
		// [unverified: prices from HF only]
		close2;
		callshop "booster_mission_shop", 1;
		end;
	}
	close;

// GN tab 1-3: one Special Currency Pouch (101453) for one Level Up
// Ticket of any band, or for 50 Booster Coins. kRO opens it after
// Episode 19, which rAthena doesn't have; the pouch itself needs
// Lv 200 (CL), so this mod opens the exchange at Lv 200.
// [unverified: one source (GN)] [deviation: Lv 200 instead of EP 19]
L_Exchange:
	if (BaseLevel < 200) {
		mes "[Lime Evenor]";
		mes "The currency exchange opens once you've finished Episode 19.";
		mes "^777777(This world has no Episode 19, so it opens at Base Lv 200.)^000000";
		close;
	}
	mes "[Lime Evenor]";
	mes "I'll swap a spare Level Up Ticket, or 50 Booster Coins, for a ^0000ffSpecial Currency Pouch^000000.";
	next;
	setarray .@pay[0], 1000989, 1000990, 1000991, 1000992, 1000993, 1000254;
	setarray .@amt[0], 1, 1, 1, 1, 1, 50;
	.@menu$ = "";
	for (.@i = 0; .@i < getarraysize(.@pay); .@i++)
		.@menu$ += .@amt[.@i] + " x " + getitemname(.@pay[.@i]) + " (have " + countitem(.@pay[.@i]) + "):";
	.@menu$ += "Cancel";
	.@s = select(.@menu$) - 1;
	if (.@s >= getarraysize(.@pay))
		close;
	if (countitem(.@pay[.@s]) < .@amt[.@s]) {
		mes "[Lime Evenor]";
		mes "You don't have enough " + getitemname(.@pay[.@s]) + ".";
		close;
	}
	if (!checkweight(101453, 1)) {
		mes "[Lime Evenor]";
		mes "You're carrying too much.";
		close;
	}
	delitem .@pay[.@s], .@amt[.@s];
	getitem 101453, 1;
	mes "[Lime Evenor]";
	mes "Here you go.";
	close;

//------------------------------------------------------------
// 2. Missions
//------------------------------------------------------------
L_Missions:
	mes "[Lime Evenor]";
	mes "Hunting requests are for Lv 1 to 80. Booster Missions open at Lv 80.";
	next;
	switch (select("Lv 1-80 hunting requests", "Booster Missions (Lv 80+)", "Other hunts", "How to find quest monsters", "Cancel")) {
	case 1: callsub L_Hunts; end;
	case 2: callsub L_MissionList; end;
	case 3: callsub L_OtherHunts; end;
	case 4: callsub L_FindMonsters; end;
	}
	close;

// kRO's menu has "other hunts" here (HF only); what it lists isn't in
// any source. This mod sends players to the Eden Group's mission
// boards instead [deviation, Steven 2026-10-17]. The warp is the Eden
// Teleport Officers' own (npc/re/quests/eden/eden_common.txt).
L_OtherHunts:
	mes "[Lime Evenor]";
	mes "Want more hunting work? My colleagues in the ^0000ffEden Group^000000 post requests for every level on their mission boards.";
	mes "Their base is hidden in Morocc. Any Eden Teleport Officer in town can take you, or I can send you now.";
	next;
	if (select("Take me to the Eden Group", "Maybe later") == 2)
		close;
	close2;
	warp "moc_para01", 31, 14;
	end;

// kRO's "how to find quest monsters" page (HF only); this text is the
// mod's own.
L_FindMonsters:
	mes "[Lime Evenor]";
	mes "Open your ^0000ffquest window^000000 and pick the request.";
	mes "It shows each monster and how many are left, and its description says where to find them.";
	next;
	mes "[Lime Evenor]";
	mes "Where the hunting requests send you:";
	mes "^0000ffLv 1^000000 Fabre: fields around Geffen and Prontera.";
	mes "^0000ffLv 11^000000 Payon's forests and fields.";
	mes "^0000ffLv 20^000000 Prontera Culvert, through the Knight recruiter in prt_in.";
	mes "^0000ffLv 35^000000 Mjolnir Dead Pit, north of Geffen.";
	mes "^0000ffLv 50^000000 Byalan Undersea Tunnel, by the sailor in Izlude.";
	mes "^0000ffLv 65^000000 Payon Cave, in the archer village.";
	mes "^0000ffLv 80^000000 Clock Tower, in Al De Baran.";
	next;
	mes "[Lime Evenor]";
	mes "Booster Missions are memorial dungeons and Illusion Investigation tasks. Their quest descriptions name the dungeon or the team member to see.";
	mes "In a memorial dungeon, party members near the boss when it falls all get the credit.";
	close;

// GN tab 1-4: one-time hunts that pay EXP (the 2021 run made them
// daily and paid coins). Quest IDs and areas: CL 18029-18041.
// Targets and EXP: GN + HF. The Mjolnir hunt's third target,
// 가이아스, is taken to be Giearth [unverified: mob].
L_Hunts:
	.@menu$ = "";
	.@n = 0;
	for (.@i = 0; .@i < getarraysize(.hunt_quest); .@i++) {
		if (BaseLevel < .hunt_lv[.@i] || isbegin_quest(.hunt_quest[.@i]) == 2)
			continue;
		.@list[.@n] = .@i;
		.@n++;
		.@state = isbegin_quest(.hunt_quest[.@i]);
		.@menu$ += "Lv " + .hunt_lv[.@i] + " " + .hunt_name$[.@i] + (.@state == 1 ? (checkquest(.hunt_quest[.@i], HUNTING) == 2 ? " ^0000ff(report)^000000" : " (in progress)") : "") + ":";
	}
	if (.@n == 0) {
		mes "[Lime Evenor]";
		mes "There are no hunting requests for you right now.";
		close;
	}
	.@menu$ += "Cancel";
	.@s = select(.@menu$) - 1;
	if (.@s >= .@n)
		close;
	.@i = .@list[.@s];
	.@q = .hunt_quest[.@i];
	mes "[Lime Evenor]";
	if (isbegin_quest(.@q) == 0) {
		mes "^0000ff" + .hunt_name$[.@i] + "^000000";
		mes .hunt_desc$[.@i];
		mes "Report back to me through your Booster Call when you're done.";
		next;
		if (select("Accept", "Not now") == 2)
			close;
		setquest .@q;
		mes "[Lime Evenor]";
		mes "Good luck!";
		close;
	}
	if (checkquest(.@q, HUNTING) != 2) {
		mes "You haven't finished hunting yet. Check your quest window.";
		close;
	}
	completequest .@q;
	getexp2 .hunt_bexp[.@i], .hunt_jexp[.@i];
	mes "Well done! Here's your reward.";
	close;

// The 22 Booster Missions (CL 18171-18192, plus 18205 for the Hard
// version of mission 1). One-time each [unverified: one-time]. Names,
// levels and quest IDs: CL + GN + HF; rewards: GN + HF.
//
// How each kind is checked on this server:
//  K (kill): the mission quest counts the boss kill (db/quest_db.yml),
//    as kRO's summary says ("<boss> 처치"). Party members within 14
//    cells of the kill get the credit too.
//  J (join): stock rAthena's Commander Joel quest 3470 is complete.
//  I (Illusion): the stock member's task leaves its 4 h standby quest
//    (or a finished hunt) in the quest log, so Lime accepts either.
//    rAthena keeps nothing once the player talks to the member again
//    after the standby, so report before that.
//  N (not available): the memorial dungeon isn't in rAthena (or the
//    app doesn't load it), so the mission is listed but can't start.
L_MissionList:
	if (BaseLevel < 80) {
		mes "[Lime Evenor]";
		mes "Booster Missions open at Base Lv 80.";
		close;
	}
	.@menu$ = "";
	.@n = 0;
	for (.@i = 0; .@i < getarraysize(.mission_lv); .@i++) {
		if (BaseLevel < .mission_lv[.@i])
			continue;
		.@q = callsub(S_MissionQuest, .@i);
		.@state = isbegin_quest(.@q);
		if (.@state == 2)
			continue;
		.@list[.@n] = .@i;
		.@n++;
		.@tag$ = "";
		if (.mission_kind$[.@i] == "N")
			.@tag$ = " ^777777(not on this server)^000000";
		else if (.@state == 1)
			.@tag$ = (callsub(S_MissionDone, .@i, .@q) ? " ^0000ff(report)^000000" : " (in progress)");
		.@menu$ += "Lv " + .mission_lv[.@i] + " " + .mission_name$[.@i] + .@tag$ + ":";
	}
	if (.@n == 0) {
		mes "[Lime Evenor]";
		mes "You've done every Booster Mission open to you. Well done!";
		close;
	}
	mes "[Lime Evenor]";
	mes "Which mission?";
	next;
	.@menu$ += "Cancel";
	.@s = select(.@menu$) - 1;
	if (.@s >= .@n)
		close;
	.@i = .@list[.@s];
	.@q = callsub(S_MissionQuest, .@i);
	.@name$ = .mission_name$[.@i];
	mes "[Lime Evenor]";
	if (.mission_kind$[.@i] == "N") {
		mes "^0000ff" + .@name$ + "^000000";
		mes "I'd send you there, but that dungeon hasn't reached this world yet.";
		mes "^777777(This server has no " + .mission_where$[.@i] + ", so this mission can't be done.)^000000";
		close;
	}
	if (isbegin_quest(.@q) == 0) {
		mes "^0000ff" + .@name$ + "^000000";
		mes .mission_desc$[.@i];
		.@t = .mission_ticket[.@i];
		mes "Reward: " + .@t + " Booster Mission Ticket" + (.@t > 1 ? "s" : "") + (.mission_item1[.@i] ? ", " + .mission_amt1[.@i] + " " + getitemname(.mission_item1[.@i]) : "") + (.mission_item2[.@i] ? ", " + .mission_amt2[.@i] + " " + getitemname(.mission_item2[.@i]) : "") + ".";
		mes "Report back to me through your Booster Call when you're done.";
		next;
		if (select("Accept", "Not now") == 2)
			close;
		setquest .@q;
		mes "[Lime Evenor]";
		mes "Good luck!";
		close;
	}
	if (!callsub(S_MissionDone, .@i, .@q)) {
		mes "^0000ff" + .@name$ + "^000000";
		mes .mission_desc$[.@i];
		mes "Come back once it's done.";
		close;
	}
	.@t = .mission_ticket[.@i];
	if (!checkweight(1000984, .@t) || (.mission_item1[.@i] && !checkweight(.mission_item1[.@i], .mission_amt1[.@i])) || (.mission_item2[.@i] && !checkweight(.mission_item2[.@i], .mission_amt2[.@i]))) {
		mes "You're carrying too much. Make some room and report again.";
		close;
	}
	completequest .@q;
	getitem 1000984, .@t;
	if (.mission_item1[.@i])
		getitem .mission_item1[.@i], .mission_amt1[.@i];
	if (.mission_item2[.@i])
		getitem .mission_item2[.@i], .mission_amt2[.@i];
	mes "Mission complete! Here's your reward.";
	close;

// The quest a mission uses. Mission 1 is Easy (18171) or Hard (18205):
// rAthena's Half Moon picks Hard when the party leader is Lv 130+, so
// a Lv 130+ player gets the Hard version unless the Easy one is open.
S_MissionQuest:
	.@i = getarg(0);
	if (.@i != 0)
		return .mission_quest[.@i];
	if (isbegin_quest(18171))
		return 18171;
	if (isbegin_quest(18205))
		return 18205;
	return (BaseLevel >= 130 ? 18205 : 18171);

// Whether an active mission is finished.
S_MissionDone:
	.@i = getarg(0);
	.@q = getarg(1);
	.@k$ = .mission_kind$[.@i];
	if (.@k$ == "K")
		return (checkquest(.@q, HUNTING) == 2);
	if (.@k$ == "J")
		return (isbegin_quest(3470) == 2);
	if (.@k$ == "I")
		return (checkquest(.mission_standby[.@i]) >= 0 || checkquest(.mission_hunt[.@i], HUNTING) == 2);
	return 0;

	mes "What would you like to enchant?";
	next;
	.@s = select("Booster Weapon (Booster Coins)", "Booster costume headgear (ticket)", "Booster costume garment (ticket)", "Cancel");
	if (.@s == 4)
		close;
	close2;
	if (.@s == 1)
		item_enchant(36);
	else if (.@s == 2)
		item_enchant(37);
	else
		item_enchant(38);
	end;

//------------------------------------------------------------
// 4. Level Up Tickets (CL + Inven 2022-06-19): +1 Base Level,
// only inside the ticket's band.
//------------------------------------------------------------
L_LevelUp:
	.@menu$ = "";
	for (.@i = 0; .@i < getarraysize(.lvt_item); .@i++)
		.@menu$ += getitemname(.lvt_item[.@i]) + " (have " + countitem(.lvt_item[.@i]) + "):";
	.@menu$ += "Cancel";
	mes "[Lime Evenor]";
	mes "Which ticket? Each one raises your Base Level by 1 inside its range.";
	next;
	.@s = select(.@menu$) - 1;
	if (.@s >= getarraysize(.lvt_item))
		close;
	mes "[Lime Evenor]";
	if (countitem(.lvt_item[.@s]) < 1) {
		mes "You don't have that ticket.";
		close;
	}
	if (BaseLevel < .lvt_min[.@s] || BaseLevel > .lvt_max[.@s]) {
		mes "That ticket only works from Base Lv " + .lvt_min[.@s] + " to " + .lvt_max[.@s] + ".";
		close;
	}
	if (NextBaseExp <= 0) {
		mes "You can't level up any further.";
		close;
	}
	delitem .lvt_item[.@s], 1;
	getexp2 NextBaseExp - BaseExp, 0;
	mes "Done! You're now Base Lv " + BaseLevel + ".";
	close;

//------------------------------------------------------------
// 5. Milestones (GN tab 1-3 + HF; flags CL 18193-18204).
// Premium rewards need the Premium Call; upgrading later still
// lets the player claim every premium reward already passed.
//------------------------------------------------------------
L_Milestones:
	.@premium = getarg(0);
	.@menu$ = "";
	for (.@i = 0; .@i < 6; .@i++) {
		.@ok = callsub(L_MilestoneReached, .@i);
		.@n_done = (isbegin_quest(18193 + .@i) == 2);
		.@p_done = (isbegin_quest(18199 + .@i) == 2);
		.@menu$ += .ms_name$[.@i] + ": " + (.@ok ? "" : "^777777not reached^000000 ") + (.@n_done ? "normal claimed" : "normal") + ", " + (.@premium ? (.@p_done ? "premium claimed" : "premium") : "premium locked") + ":";
	}
	.@menu$ += "Cancel";
	mes "[Lime Evenor]";
	mes "Which reward would you like?";
	next;
	.@s = select(.@menu$) - 1;
	if (.@s >= 6)
		close;
	mes "[Lime Evenor]";
	if (!callsub(L_MilestoneReached, .@s)) {
		mes "You haven't reached that milestone yet.";
		if (.@s == 3)
			mes "^777777(This world has no Episode 19, so this one opens at Base Lv 210.)^000000";
		close;
	}
	.@paid = 0;
	if (isbegin_quest(18193 + .@s) != 2) {
		callsub L_MilestoneNormal, .@s;
		setquest 18193 + .@s;
		completequest 18193 + .@s;
		.@paid++;
	}
	if (.@premium && isbegin_quest(18199 + .@s) != 2) {
		callsub L_MilestonePremium, .@s;
		setquest 18199 + .@s;
		completequest 18199 + .@s;
		.@paid++;
	}
	if (.@paid)
		mes "Congratulations! Here are your rewards for " + .ms_name$[.@s] + ".";
	else
		mes "You've already claimed " + (.@premium ? "both rewards" : "this reward") + ".";
	close;

// Episode completion values are the last step of each main quest in
// rAthena: terra_gloria_main 26 (17.1), ep17_2_main 36, ep18_main 57.
// rAthena has no Episode 19, so it opens at Base Lv 210 (kRO's EP 19
// level in GN). [deviation]
L_MilestoneReached:
	switch (getarg(0)) {
	case 0: return (terra_gloria_main >= 26);
	case 1: return (ep17_2_main >= 36);
	case 2: return (ep18_main >= 57);
	case 3: return (BaseLevel >= 210);
	case 4: return (BaseLevel >= 230);
	case 5: return (BaseLevel >= 250);
	}
	return 0;

L_MilestoneNormal:
	switch (getarg(0)) {
	case 0:
		getitem 1000994, 5;
		getexp2 8056742, 0;
		break;
	case 1:
		getexp2 18517460, 0;
		break;
	case 2:
		getexp2 844635696, 0;
		break;
	case 3:
		getexp2 2604376656, 0;
		break;
	case 4:
		getitem 101476, 1;
		break;
	case 5:
		getitem 101460, 1;
		getitem 1000984, 10;
		break;
	}
	return;

L_MilestonePremium:
	switch (getarg(0)) {
	case 0:
		getitem 25723, 80;	// Cor Core
		getitem 25669, 50;	// Mysterious Component
		break;
	case 1:
		getitem 1000985, 5;
		getitem 1000103, 110;
		break;
	case 2:
		getitem 1000405, 660;
		break;
	case 3:
		getitem 1000608, 370;	// 설화의 꽃잎 [unverified: that this is the rAthena item]
		break;
	case 4:
		getitem 101460, 1;
		break;
	case 5:
		getitem 23919, 50;	// [unverified: K_Secret_Key vs the [Reward] key 102701]
		break;
	}
	return;

//------------------------------------------------------------
// 7. Premium upgrade (GN tab 1-3 + HF): Booster Call + 150
// Nyangvine Fruit. Pays 10 Status and 10 Skill Reset Tickets, a
// Boarding Rein Box and a Premium Costume Enchant Ticket at once.
//------------------------------------------------------------
L_Upgrade:
	// kRO cost: 150 Nyangvine Fruit (6909), a cash-shop currency with
	// no source in rAthena. This mod charges premium_upgrade_coins
	// Booster Coins instead (default 50) [deviation, mod setting];
	// setting it to 0 restores kRO's Nyangvine Fruit cost.
	.@coins = callfunc("F_ModSetting", "mega-booster", "premium_upgrade_coins", 50);
	.@cost = (.@coins > 0 ? 1000254 : 6909);
	.@amount = (.@coins > 0 ? .@coins : 150);
	mes "[Lime Evenor]";
	mes "Hand me your Booster Call and ^0000ff" + .@amount + " " + getitemname(.@cost) + "^000000 and I'll give you a Premium Booster Call.";
	mes "You also get 10 Status and 10 Skill Reset Tickets, a Boarding Rein Box and a Premium Costume Enchant Ticket.";
	mes "Premium rewards for milestones you've already passed can still be claimed afterwards.";
	next;
	if (select("Upgrade", "Not now") == 2)
		close;
	mes "[Lime Evenor]";
	if (countitem(100337) < 1 || countitem(.@cost) < .@amount) {
		mes "You need the Booster Call and " + .@amount + " " + getitemname(.@cost) + ".";
		close;
	}
	if (!checkweight(101473, 1, 1000967, 10, 1000983, 10, 101589, 1, 1000995, 1)) {
		mes "You're carrying too much.";
		close;
	}
	delitem 100337, 1;
	delitem .@cost, .@amount;
	getitem 101473, 1;
	getitem 1000967, 10;
	getitem 1000983, 10;
	getitem 101589, 1;
	getitem 1000995, 1;
	mes "Your Booster Call is now a Premium Booster Call!";
	close;

OnInit:
	// Hunts: CL quest IDs, GN targets and EXP.
	setarray .hunt_quest[0], 18029, 18031, 18033, 18035, 18037, 18039, 18041;
	setarray .hunt_lv[0], 1, 11, 20, 35, 50, 65, 80;

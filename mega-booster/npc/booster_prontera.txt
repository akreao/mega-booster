// True when the player holds either Booster Call.
function	script	F_Booster_Holder	{
	return (countitem(100337) > 0 || countitem(101473) > 0);
}

// Raises Base and Job Level to at least the given levels, one level
// at a time (getexp2 ignores rates). Stops early at a level cap.
function	script	F_Booster_LevelTo	{
	.@base = getarg(0);
	.@job = getarg(1);
	while (BaseLevel < .@base && NextBaseExp > 0) {
		.@lv = BaseLevel;
		getexp2 NextBaseExp - BaseExp, 0;
		if (BaseLevel == .@lv)
			break;
	}
	while (JobLevel < .@job && NextJobExp > 0) {
		.@lv = JobLevel;
		getexp2 0, NextJobExp - JobExp;
		if (JobLevel == .@lv)
			break;
	}
	return;
}

// Job-change gift: what the official job quest gives for the same
// change, handed out by the Job Agent right after a normal (not
// transcended) change while the job_quest_rewards setting is on.
// Transcended changes give nothing, as at Valhalla and Juno. Opens a
// new dialog page to show what was given;
// getarg(0) is the new job.
// The agent asks for Job Lv 50 before a 2nd job, so the 2nd jobs get
// the quests' Job Lv 50 gifts.
function	script	F_Booster_JobGift	{
	switch (getarg(0)) {
	// 1st jobs: rAthena's renewal job quests (npc/re/jobs/1-1). The kRO
	// client has these Novice items; iRO's job quests give none.
	// [unverified: kRO gifts, rAthena renewal scripts only]
	case Job_Swordman:	setarray .@item[0], 13415; setarray .@amt[0], 1; break;	// N_Falchion
	case Job_Mage:		setarray .@item[0], 1639; setarray .@amt[0], 1; break;	// N_Rod
	case Job_Archer:	setarray .@item[0], 1742, 12004, 12009, 12008; setarray .@amt[0], 1, 1, 1, 1; break;	// N_Composite_Bow, Arrow_Container, Silver_Arrow_Container, Fire_Arrow_Container
	case Job_Acolyte:	setarray .@item[0], 1545; setarray .@amt[0], 1; break;	// N_Mace
	case Job_Merchant:	setarray .@item[0], 1381; setarray .@amt[0], 1; break;	// N_Battle_Axe
	case Job_Thief:		setarray .@item[0], 13041; setarray .@amt[0], 1; break;	// N_Main_Gauche
	// Expanded 1st jobs and Super Novice: rAthena's job quests and
	// iRO wiki agree.
	case Job_Taekwon:	setarray .@item[0], 2101; setarray .@amt[0], 1; break;	// Guard
	case Job_Ninja:		setarray .@item[0], 13010; setarray .@amt[0], 1; break;	// Asura
	case Job_Gunslinger:	setarray .@item[0], 13180, 12149, 12151; setarray .@amt[0], 1, 2, 1; break;	// Novice_Rifle, Bullet_Case, Bullet_Case_Silver
	case Job_Super_Novice:	setarray .@item[0], 2339; setarray .@amt[0], 1; break;	// G_Strings
	// 2nd jobs at Job Lv 50: rAthena's job quests and iRO wiki agree.
	// The Sage and Bard quests give nothing on the change (the Bard's
	// instrument is a trade for 60 trunks).
	case Job_Knight:	setarray .@item[0], 656; setarray .@amt[0], 7; break;	// Awakening_Potion
	case Job_Priest:	setarray .@item[0], 1551; setarray .@amt[0], 1; break;	// Bible
	case Job_Wizard:	setarray .@item[0], 505; setarray .@amt[0], 6; break;	// Blue_Potion
	case Job_Blacksmith:	setarray .@item[0], 999; setarray .@amt[0], 30; break;	// Steel
	case Job_Hunter:	setarray .@item[0], 1718; setarray .@amt[0], 1; break;	// Hunter_Bow
	case Job_Crusader:	setarray .@item[0], 504; setarray .@amt[0], 12; break;	// White_Potion
	case Job_Monk:		setarray .@item[0], 1804; setarray .@amt[0], 1; break;	// Knuckle_Duster_
	case Job_Alchemist:	setarray .@item[0], 7133; setarray .@amt[0], 1; break;	// Slim_Potion_Create_Book
	case Job_Rogue:		setarray .@item[0], 1220; setarray .@amt[0], 1; break;	// Gladius_
	case Job_Dancer:	setarray .@item[0], 1953; setarray .@amt[0], 1; break;	// Line_
	case Job_Assassin:
		// The quest lets a Job Lv 49+ Thief pick one.
		setarray .@pick[0], 1251, 1253, 1208, 1220;	// Jur_, Katar_, Main_Gauche_, Gladius_
		next;
		mes "[Mega Booster Job Agent]";
		mes "The Assassin job quest lets you pick a weapon as your gift. Which one would you like?";
		next;
		.@s = select(getitemname(.@pick[0]), getitemname(.@pick[1]), getitemname(.@pick[2]), getitemname(.@pick[3])) - 1;
		setarray .@item[0], .@pick[.@s]; setarray .@amt[0], 1;
		.@picked = 1;
		break;
	// Druid: rAthena has no Druid job quest and the kRO client lists no
	// gift. Mod choice: the Novice weapon the Druid line can equip, as
	// Booster Pack (1) gives them a mace. [unverified]
	case JOB_DRUID:		setarray .@item[0], 1545; setarray .@amt[0], 1; break;	// N_Mace
	default:
		return;
	}
	if (!.@picked) {
		next;
		mes "[Mega Booster Job Agent]";
	}
	mes "And here is the gift the job quest would have given you:";
	for (.@i = 0; .@i < getarraysize(.@item); .@i++) {
		getitem .@item[.@i], .@amt[.@i];
		mes "- " + (.@amt[.@i] > 1 ? .@amt[.@i] + " x " : "") + getitemname(.@item[.@i]);
	}
	return;
}

// Job quest skills: the skills the official skill quests teach a job
// line (rAthena npc/quests/skills, renewal's Decorate Cart quest, and
// the IsQuest skills in rAthena's skill_db; iRO wiki's Quest Skills
// list has the same set). The Job Agent teaches the whole line up to
// the new job, so a 2nd job also gets its 1st job's skills, and a
// transcended job gets them back after rebirth, as Valhalla does.
// Expanded jobs and the Druid line have no skill quests. getarg(0) is
// the new job. Returns the names of the skills newly learned, or "".
function	script	F_Booster_QuestSkills	{
	.@eac = eaclass(getarg(0));
	switch (.@eac & EAJ_BASEMASK) {
	case EAJ_SWORDMAN:
		setarray .@sk$[0], "SM_MOVINGRECOVERY", "SM_FATALBLOW", "SM_AUTOBERSERK";
		setarray .@nm$[0], "HP Recovery While Moving", "Fatal Blow", "Berserk";
		break;
	case EAJ_MAGE:
		setarray .@sk$[0], "MG_ENERGYCOAT";
		setarray .@nm$[0], "Energy Coat";
		break;
	case EAJ_ARCHER:
		setarray .@sk$[0], "AC_MAKINGARROW", "AC_CHARGEARROW";
		setarray .@nm$[0], "Arrow Crafting", "Arrow Repel";
		break;
	case EAJ_ACOLYTE:
		setarray .@sk$[0], "AL_HOLYLIGHT";
		setarray .@nm$[0], "Holy Light";
		break;
	case EAJ_MERCHANT:
		setarray .@sk$[0], "MC_CARTREVOLUTION", "MC_CHANGECART", "MC_LOUD", "MC_CARTDECORATE";
		setarray .@nm$[0], "Cart Revolution", "Change Cart", "Crazy Uproar", "Cart Decoration";
		break;
	case EAJ_THIEF:
		setarray .@sk$[0], "TF_SPRINKLESAND", "TF_BACKSLIDING", "TF_PICKSTONE", "TF_THROWSTONE";
		setarray .@nm$[0], "Sand Attack", "Back Slide", "Find Stone", "Stone Fling";
		break;
	default:
		return "";
	}
	.@n = getarraysize(.@sk$);
	if (.@eac & EAJL_2) {
		// The matching normal 2nd job, so transcended jobs share the list.
		switch (roclass(.@eac & ~(EAJL_UPPER | EAJL_BABY | EAJL_THIRD), Sex)) {
		case Job_Knight:	setarray .@sk$[.@n], "KN_CHARGEATK"; setarray .@nm$[.@n], "Charge Attack"; break;
		case Job_Crusader:	setarray .@sk$[.@n], "CR_SHRINK"; setarray .@nm$[.@n], "Shrink"; break;
		case Job_Wizard:	setarray .@sk$[.@n], "WZ_SIGHTBLASTER"; setarray .@nm$[.@n], "Sight Blaster"; break;
		case Job_Sage:		setarray .@sk$[.@n], "SA_CREATECON"; setarray .@nm$[.@n], "Create Elemental Converter"; .@sage = 1; break;
		case Job_Hunter:	setarray .@sk$[.@n], "HT_PHANTASMIC"; setarray .@nm$[.@n], "Phantasmic Arrow"; break;
		case Job_Bard:		setarray .@sk$[.@n], "BA_PANGVOICE"; setarray .@nm$[.@n], "Pang Voice"; break;
		case Job_Dancer:	setarray .@sk$[.@n], "DC_WINKCHARM"; setarray .@nm$[.@n], "Charming Wink"; break;
		case Job_Priest:	setarray .@sk$[.@n], "PR_REDEMPTIO"; setarray .@nm$[.@n], "Redemptio"; break;
		case Job_Monk:		setarray .@sk$[.@n], "MO_KITRANSLATION", "MO_BALKYOUNG"; setarray .@nm$[.@n], "Spiritual Bestowment", "Excruciating Palm"; break;
		case Job_Blacksmith:	setarray .@sk$[.@n], "BS_UNFAIRLYTRICK", "BS_GREED"; setarray .@nm$[.@n], "Dubious Salesmanship", "Greed"; break;
		case Job_Alchemist:	setarray .@sk$[.@n], "AM_BIOETHICS"; setarray .@nm$[.@n], "Bioethics"; break;
		case Job_Assassin:	setarray .@sk$[.@n], "AS_SONICACCEL", "AS_VENOMKNIFE"; setarray .@nm$[.@n], "Sonic Acceleration", "Venom Knife"; break;
		case Job_Rogue:		setarray .@sk$[.@n], "RG_CLOSECONFINE"; setarray .@nm$[.@n], "Close Confine"; break;
		}
	}
	.@learned$ = "";
	for (.@i = 0; .@i < getarraysize(.@sk$); .@i++) {
		if (getskilllv(.@sk$[.@i]) > 0)
			continue;
		skill .@sk$[.@i], 1, SKILL_PERM;
		.@learned$ += (.@learned$ == "" ? "" : ", ") + .@nm$[.@i];
	}
	// The Sage quest teaches one Elemental Change of the player's choice.
	if (.@sage && getskilllv("SA_ELEMENTWATER") + getskilllv("SA_ELEMENTGROUND") + getskilllv("SA_ELEMENTFIRE") + getskilllv("SA_ELEMENTWIND") == 0) {
		setarray .@el$[0], "SA_ELEMENTWATER", "SA_ELEMENTGROUND", "SA_ELEMENTFIRE", "SA_ELEMENTWIND";
		setarray .@eln$[0], "Elemental Change (Water)", "Elemental Change (Earth)", "Elemental Change (Fire)", "Elemental Change (Wind)";
		next;
		mes "[Mega Booster Job Agent]";
		mes "The Sage skill quest teaches one Elemental Change. Which element would you like?";
		next;
		.@s = select(.@eln$[0], .@eln$[1], .@eln$[2], .@eln$[3]) - 1;
		skill .@el$[.@s], 1, SKILL_PERM;
		.@learned$ += (.@learned$ == "" ? "" : ", ") + .@eln$[.@s];
	}
	return .@learned$;
}

// Pick-one box: getarg(0) is the box, then item, amount pairs.
// Used by the item scripts in db/item_db.yml. The boxes are
// DelayConsume, so the script deletes the box itself, and only once
// the player has chosen.
function	script	F_Booster_Pick	{
	.@box = getarg(0);
	.@n = (getargcount() - 1) / 2;
	for (.@i = 0; .@i < .@n; .@i++) {
		.@item[.@i] = getarg(1 + .@i * 2);
		.@amt[.@i] = getarg(2 + .@i * 2);
		.@menu$ += (.@amt[.@i] > 1 ? .@amt[.@i] + " x " : "") + getitemname(.@item[.@i]) + ":";
	}
	.@menu$ += "Cancel";
	mes "[" + getitemname(.@box) + "]";
	mes "Choose one.";
	.@s = select(.@menu$) - 1;
	if (.@s >= .@n)
		close;
	if (countitem(.@box) < 1)
		close;
	if (!checkweight(.@item[.@s], .@amt[.@s])) {
		mes "You're carrying too much.";
		close;
	}
	delitem .@box, 1;
	getitem .@item[.@s], .@amt[.@s];
	close;
}

//------------------------------------------------------------
// Mega Booster Job Agent
// GN + HF: free rebirth at Lv 99; after the transcended 1st job the
// agent sets Base 60 / Job 50, after the transcended 2nd job Base 99
// / Job 70. The job-change requirements (1st job at Job Lv 10, Super
// Novice also Base 45, 2nd job at Job Lv 50; 3rd and 4th jobs through
// the normal quests) are HF only. [unverified: requirements]
// The Druid line (Druid from Novice, then Karnos and Alitea) is this
// mod's own addition; see the Druid branch below. It is off on clients
// without the Druid line's art (F_Booster_NoDruid, booster_client.txt).
//------------------------------------------------------------
prt_in,34,95,0	script	Mega Booster Job Agent#bst	10210,{
	mes "[Mega Booster Job Agent]";
	if (!callfunc("F_Booster_Holder")) {
		mes "I only help adventurers who carry a Booster Call.";
		close;
	}
	.@eac = eaclass();
	.@base = .@eac & EAJ_BASEMASK;
	.@basic = (.@base == EAJ_SWORDMAN || .@base == EAJ_MAGE || .@base == EAJ_ARCHER || .@base == EAJ_ACOLYTE || .@base == EAJ_MERCHANT || .@base == EAJ_THIEF);

	// Novice: first job.
	if (Class == Job_Novice) {
		if (JobLevel < 10) {
			mes "Come back at Job Lv 10 and I'll get you your first job.";
			close;
		}
		setarray .@jobs[0], Job_Swordman, Job_Mage, Job_Archer, Job_Acolyte, Job_Merchant, Job_Thief, Job_Taekwon, Job_Gunslinger, Job_Ninja, JOB_DRUID, Job_Super_Novice;
		// No Druid on a client without the Druid line's art.
		if (callfunc("F_Booster_NoDruid"))
			deletearray .@jobs[9], 1;
		.@menu$ = "";
		for (.@i = 0; .@i < getarraysize(.@jobs); .@i++)
			.@menu$ += jobname(.@jobs[.@i]) + ":";
		mes "Which job would you like?";
		next;
		.@s = select(.@menu$ + "Cancel") - 1;
		if (.@s >= getarraysize(.@jobs))
			close;
		if (.@jobs[.@s] == Job_Super_Novice && BaseLevel < 45) {
			mes "[Mega Booster Job Agent]";
			mes "Super Novice needs Base Lv 45.";
			close;
		}
		callsub L_Change, .@jobs[.@s], 0, 0;
		end;
	}

	// High Novice: transcended 1st job, then Base 60 / Job 50.
	if (Class == Job_Novice_High) {
		if (JobLevel < 10) {
			mes "Come back at Job Lv 10.";
			close;
		}
		if (lastJob)
			.@to = roclass((eaclass(lastJob) & EAJ_BASEMASK) | EAJL_UPPER);
		else {
			setarray .@jobs[0], Job_Swordman_High, Job_Mage_High, Job_Archer_High, Job_Acolyte_High, Job_Merchant_High, Job_Thief_High;
			.@menu$ = "";
			for (.@i = 0; .@i < getarraysize(.@jobs); .@i++)
				.@menu$ += jobname(.@jobs[.@i]) + ":";
			.@s = select(.@menu$ + "Cancel") - 1;
			if (.@s >= getarraysize(.@jobs))
				close;
			.@to = .@jobs[.@s];
		}
		mes "You'll become " + callfunc("F_InsertArticle", jobname(.@to)) + ", and I'll raise you to Base Lv 60 / Job Lv 50.";
		next;
		if (select("Go ahead", "Not now") == 2)
			close;
		callsub L_Change, .@to, 60, 50;
		end;
	}

	// 1st job of the six basic classes: 2nd job (transcended: the
	// matching one, then Base 99 / Job 70).
	if (.@basic && !(.@eac & EAJL_2) && !(.@eac & EAJL_THIRD)) {
		if (JobLevel < 50) {
			mes "Come back at Job Lv 50 for your second job.";
			close;
		}
		if ((.@eac & EAJL_UPPER) && lastJob) {
			.@to = lastJob + Job_Novice_High;
			mes "You'll become " + callfunc("F_InsertArticle", jobname(.@to)) + ", and I'll raise you to Base Lv 99 / Job Lv 70.";
			next;
			if (select("Go ahead", "Not now") == 2)
				close;
			callsub L_Change, .@to, 99, 70;
			end;
		}
		.@c1 = roclass(.@eac | EAJL_2_1);
		.@c2 = roclass(.@eac | EAJL_2_2);
		mes "Which second job?";
		next;
		.@s = select(jobname(.@c1), jobname(.@c2), "Cancel");
		if (.@s == 3)
			close;
		callsub L_Change, (.@s == 1 ? .@c1 : .@c2), 0, 0;
		end;
	}

	// 2nd job of the six basic classes: free rebirth at Lv 99.
	if (.@basic && (.@eac & EAJL_2) && !(.@eac & EAJL_UPPER) && !(.@eac & EAJL_THIRD)) {
		if (BaseLevel < 99 || JobLevel < 50) {
			mes "At Base Lv 99 and Job Lv 50 I can rebirth you for free.";
			close;
		}
		if (checkcart() || checkfalcon() || checkriding() || checkdragon() || checkwug() || checkmadogear() || ismounting()) {
			mes "Please put away your cart, falcon, warg or mount before rebirth.";
			close;
		}
		if (SkillPoint) {
			mes "Please use up your skill points first.";
			close;
		}
		mes "Rebirth makes you a High Novice at Base Lv 1. Go ahead?";
		next;
		if (select("Rebirth", "Not now") == 2)
			close;
		lastJob = Class;
		jobchange Job_Novice_High;
		resetlvl 1;
		specialeffect2 EF_ANGEL2;
		mes "[Mega Booster Job Agent]";
		mes "Welcome back, High Novice!";
		close;
	}

	// Druid line: rAthena has no Druid, Karnos or Alitea job quests, so
	// the agent does those changes. Mod choice, not kRO's: Karnos at Base
	// 99 / Job 70 as the server's Rebellion and Kagerou/Oboro quests ask,
	// Alitea at Base 200 / Job 70 as the trait jobs do. [unverified]
	if (.@base == EAJ_DRUID && !(.@eac & EAJL_THIRD)) {
		if (callfunc("F_Booster_NoDruid")) {
			mes "This server's game client has no Druid, Karnos or Alitea art, so I can't change Druid jobs here.";
			close;
		}
		.@to = ((.@eac & EAJL_2_1) ? JOB_ALITEA : JOB_KARNOS);
		.@bl = ((.@to == JOB_ALITEA) ? 200 : 99);
		if ((.@eac & EAJL_BABY) && .@to == JOB_ALITEA) {
			mes "I can't take a Baby Karnos any further.";
			close;
		}
		if (.@eac & EAJL_BABY)
			.@to = JOB_BABY_KARNOS;
		if (BaseLevel < .@bl || JobLevel < 70) {
			mes "At Base Lv " + .@bl + " and Job Lv 70 I can make you " + callfunc("F_InsertArticle", jobname(.@to)) + ".";
			close;
		}
		mes "You'll become " + callfunc("F_InsertArticle", jobname(.@to)) + ". Go ahead?";
		next;
		if (select("Go ahead", "Not now") == 2)
			close;
		callsub L_Change, .@to, 0, 0;
		end;
	}

	mes "Third and fourth jobs, and expanded classes, go through the usual job quests.";
	close;

// getarg(0) job, getarg(1) base level, getarg(2) job level to raise to.
L_Change:
	if (SkillPoint) {
		mes "[Mega Booster Job Agent]";
		mes "Please use up your skill points first.";
		close;
	}
	jobchange getarg(0);
	if (getarg(1) > 0)
		callfunc "F_Booster_LevelTo", getarg(1), getarg(2);
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_ELECTRIC;
	mes "[Mega Booster Job Agent]";
	mes "Congratulations, you're now " + callfunc("F_InsertArticle", jobname(Class)) + "!";
	// Setting job_quest_rewards (on by default): normal changes get the
	// job quest's gift, transcended ones don't; every change teaches the
	// line's quest skills.
	if (callfunc("F_ModSetting", "mega-booster", "job_quest_rewards", 1)) {
		if (getarg(1) == 0)
			callfunc "F_Booster_JobGift", Class;
		.@learned$ = callfunc("F_Booster_QuestSkills", Class);
		if (.@learned$ != "") {
			next;
			mes "[Mega Booster Job Agent]";
			mes "I've also taught you the skills your job's skill quests teach: " + .@learned$ + ".";
		}
	}
	close;
}

//------------------------------------------------------------
// Reset Service: GN says one ticket each. The weight and mount
// rules are HF only. [unverified: weight <= 1500, no cart, falcon,
// warg, dragon or Mado]
//------------------------------------------------------------
prt_in,37,95,0	script	Reset Service#bst	4_F_KAFRA1,{
	mes "[Reset Service]";
	if (!callfunc("F_Booster_Holder")) {
		mes "This service is for Booster Call holders.";
		close;
	}
	mes "One Status Reset Ticket resets your stats; one Skill Reset Ticket resets your skills.";
	next;
	.@s = select("Reset stats (" + countitem(1000967) + " tickets)", "Reset skills (" + countitem(1000983) + " tickets)", "Cancel");
	if (.@s == 3)
		close;
	mes "[Reset Service]";
	if (Weight / 10 > 1500 || checkcart() || checkfalcon() || checkriding() || checkdragon() || checkwug() || checkmadogear() || ismounting()) {
		mes "Please get your weight down to 1500 and put away your cart, falcon, warg or mount first.";
		close;
	}
	.@ticket = (.@s == 1 ? 1000967 : 1000983);
	if (countitem(.@ticket) < 1) {
		mes "You need a " + getitemname(.@ticket) + ".";
		close;
	}
	delitem .@ticket, 1;
	if (.@s == 1)
		resetstatus;
	else
		resetskill;
	mes "Done!";
	close;
}

//------------------------------------------------------------
// Booster Exchanger
// CL voucher text + GN: Booster Weapons for a Booster Weapon Voucher,
// bound Illusion gear for an Illusion Equipment Voucher, bound
// Automatic gear for an Automatic Equipment Voucher, and Booster
// shadow gear. GN: episode gear only after claiming that episode's
// reward through the Booster Call (flags 18193 / 18194).
//------------------------------------------------------------
prt_in,40,95,0	script	Booster Exchanger#bst	495,{
	mes "[Booster Exchanger]";
	if (!callfunc("F_Booster_Holder")) {
		mes "I only trade with Booster Call holders.";
		close;
	}
	mes "What would you like?";
	next;
	switch (select("Booster Weapon (Booster Weapon Voucher)", "Illusion gear (Illusion Equipment Voucher)", "Automatic gear (Automatic Equipment Voucher)", "Booster shadow gear (15 Booster Coins)", "Illusion voucher or upgrade package (Booster Coins)", "Cancel")) {
	case 1:
		.@shop$ = "booster_ex_weapon";
		break;
	case 2:
		if (BaseLevel < 130 || isbegin_quest(18193) != 2) {
			mes "[Booster Exchanger]";
			mes "Illusion gear needs Base Lv 130 and your Episode 17.1 reward from Lime.";
			close;
		}
		.@shop$ = "booster_ex_illusion";
		break;
	case 3:
		if (BaseLevel < 160 || isbegin_quest(18194) != 2) {
			mes "[Booster Exchanger]";
			mes "Automatic gear needs Base Lv 160 and your Episode 17.2 reward from Lime.";
			close;
		}
		.@shop$ = "booster_ex_automatic";
		break;
	case 4:
		// [unverified: Lv 100, both HF pages by one author]
		if (BaseLevel < 100) {
			mes "[Booster Exchanger]";
			mes "Booster shadow gear needs Base Lv 100.";
			close;
		}
		.@shop$ = "booster_ex_shadow";
		break;
	case 5:
		.@shop$ = "booster_ex_coins";
		break;
	default:
		close;
	}
	// The trades are barter shops in db/barters.yml; prices are there.
	close2;
	callshop .@shop$, 1;
	end;
}
